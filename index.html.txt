    <script>
        // Initialize Lucide Icons
        document.addEventListener('DOMContentLoaded', () => {
            lucide.createIcons();
            handleInitialRoute();
        });

        // SPA Navigation Manager
        function navigateTo(viewId) {
            // Hide all pages
            const pages = document.querySelectorAll('.page-section');
            pages.forEach(page => page.classList.remove('active'));

            // Show selected view
            const targetView = document.getElementById(`view-${viewId}`);
            if (targetView) {
                targetView.classList.add('active');
                window.scrollTo({ top: 0, behavior: 'smooth' });
            }

            // Update Nav link highlights
            const navLinks = document.querySelectorAll('.nav-link');
            navLinks.forEach(link => link.classList.remove('active'));
            const activeNav = document.getElementById(`nav-${viewId}`);
            if (activeNav) {
                activeNav.classList.add('active');
            }

            // Update URL Hash without reload
            history.pushState(null, null, `#${viewId}`);
        }

        // Handle Hash on Load
        function handleInitialRoute() {
            const hash = window.location.hash.replace('#', '');
            if (hash === 'sourcing-request') {
                navigateTo('home');
                openSourcingModal();
            } else if (['services', 'industries', 'how-it-works', 'about', 'insights'].includes(hash)) {
                navigateTo(hash);
            } else {
                navigateTo('home');
            }
        }

        // Listen for browser back/forward
        window.addEventListener('popstate', () => {
            handleInitialRoute();
        });

        // Mobile Menu Toggle
        function toggleMobileMenu() {
            const menu = document.getElementById('mobile-menu');
            menu.classList.toggle('hidden');
        }

        // Services Tab Switcher
        function switchTab(section, tabId) {
            // Hide all tab contents
            const contents = document.querySelectorAll('.tab-content');
            contents.forEach(content => content.classList.add('hidden'));

            // Deactivate all buttons
            const buttons = document.querySelectorAll('.tab-btn');
            buttons.forEach(btn => btn.classList.remove('active'));

            // Show selected tab & active button
            const selectedTab = document.getElementById(tabId);
            if (selectedTab) {
                selectedTab.classList.remove('hidden');
            }
            const activeBtn = document.getElementById(`tab-btn-${tabId}`);
            if (activeBtn) {
                activeBtn.classList.add('active');
            }
        }

        // FAQ Accordion Toggle
        function toggleFaq(faqId) {
            const content = document.getElementById(faqId);
            const icon = document.getElementById(`icon-${faqId}`);
            if (content.classList.contains('hidden')) {
                content.classList.remove('hidden');
                icon.style.transform = 'rotate(180deg)';
            } else {
                content.classList.add('hidden');
                icon.style.transform = 'rotate(0deg)';
            }
        }

        // Sourcing Request Modal Functions
        function openSourcingModal() {
            const modal = document.getElementById('sourcing-modal');
            modal.classList.remove('hidden');
            document.body.style.overflow = 'hidden';
            history.pushState(null, null, '#sourcing-request');
        }

        function closeSourcingModal() {
            const modal = document.getElementById('sourcing-modal');
            modal.classList.add('hidden');
            document.body.style.overflow = 'auto';
            
            // Reset status message
            const status = document.getElementById('form-status');
            status.classList.add('hidden');
            status.className = 'hidden p-3 rounded-lg text-xs font-semibold';
        }

        // Form Submit Handler
        function handleFormSubmit(event) {
            event.preventDefault();
            const status = document.getElementById('form-status');
            
            status.classList.remove('hidden');
            status.className = 'p-3 rounded-lg text-xs font-semibold bg-emerald-100 text-emerald-800 border border-emerald-300';
            status.innerHTML = '✓ Sourcing inquiry submitted successfully. Our engineering manager will contact you within 24 hours at service@sourden.com.';

            setTimeout(() => {
                event.target.reset();
                closeSourcingModal();
            }, 3000);
        }
    </script>
